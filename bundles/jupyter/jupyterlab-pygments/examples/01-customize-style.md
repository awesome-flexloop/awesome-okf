# 自定义语法高亮样式

本示例展示如何扩展和自定义 JupyterStyle，包括添加更多 token 映射、使用未激活的 CSS 变量、以及重新构建扩展。

## 示例 1：扩展 JupyterStyle 添加更多 Token 颜色

默认的 JupyterStyle 只映射了最核心的 8 个 token 类型（Text, Error, Comment, Keyword, Operator, String, Number, Punctuation），许多子 token 类型都继承父样式。以下示例创建一个更精细的样式：

```python
"""扩展 JupyterStyle，为函数名、内置函数、装饰器等添加专门颜色"""

from jupyterlab_pygments import JupyterStyle
from pygments.token import (
    Comment, Error, Generic, Keyword, Literal, Name, Number,
    Operator, Other, Punctuation, String, Text, Whitespace
)


class ExtendedJupyterStyle(JupyterStyle):
    """扩展的 JupyterStyle，利用更多 JupyterLab CSS 变量"""

    default_style = ''
    background_color = 'var(--jp-cell-editor-background)'
    highlight_color = 'var(--jp-cell-editor-active-background)'

    styles = {
        # === 保留父类的核心映射 ===
        Text:                      'var(--jp-mirror-editor-variable-color)',
        Whitespace:                '',
        Error:                     'var(--jp-mirror-editor-error-color)',
        Other:                     '',
        Comment:                   'italic var(--jp-mirror-editor-comment-color)',
        Keyword:                   'bold var(--jp-mirror-editor-keyword-color)',
        Operator:                  'bold var(--jp-mirror-editor-operator-color)',
        Operator.Word:             '',
        Literal:                   '',
        Literal.Date:              '',
        String:                    'var(--jp-mirror-editor-string-color)',
        Number:                    'var(--jp-mirror-editor-number-color)',
        Generic:                   '',
        Punctuation:               'var(--jp-mirror-editor-punctuation-color)',

        # === 新增的精细映射 ===
        # 函数名使用定义色（def color）
        Name.Function:             'var(--jp-mirror-editor-def-color)',
        # 类名使用定义色
        Name.Class:                'bold var(--jp-mirror-editor-def-color)',
        # 内置函数使用 builtin 色
        Name.Builtin:              'var(--jp-mirror-editor-builtin-color)',
        Name.Builtin.Pseudo:       'var(--jp-mirror-editor-builtin-color)',
        # 装饰器使用 meta 色
        Name.Decorator:            'var(--jp-mirror-editor-meta-color)',
        # 属性名使用 property 色
        Name.Attribute:            'var(--jp-mirror-editor-property-color)',
        # 标签使用 tag 色
        Name.Tag:                  'var(--jp-mirror-editor-tag-color)',
        # 异常使用 error 色
        Name.Exception:            'bold var(--jp-mirror-editor-error-color)',
        # 常量使用 atom 色
        Name.Constant:             'var(--jp-mirror-editor-atom-color)',
    }
```

这个扩展样式利用了 JupyterStyle 类文档中列出但未使用的 CSS 变量：
- `--jp-mirror-editor-def-color`：函数/类定义名
- `--jp-mirror-editor-builtin-color`：内置名称
- `--jp-mirror-editor-meta-color`：元信息（装饰器）
- `--jp-mirror-editor-property-color`：属性
- `--jp-mirror-editor-tag-color`：标签
- `--jp-mirror-editor-atom-color`：原子值
- `--jp-mirror-editor-error-color`：异常（加粗）

## 示例 2：修改 generate_css.py 使用自定义样式

要使用自定义的 ExtendedJupyterStyle，需要修改 `generate_css.py` 脚本：

```python
"""修改后的 generate_css.py，使用 ExtendedJupyterStyle"""
from pathlib import Path

# 从自定义模块导入
# from my_package import ExtendedJupyterStyle as JupyterStyle
from jupyterlab_pygments import JupyterStyle  # 默认样式

from pygments.formatters import HtmlFormatter

prefix = """
/*-----------------------------------------------------------------------------
| Copyright (c) Jupyter Development Team.
| Distributed under the terms of the Modified BSD License.
|----------------------------------------------------------------------------*/
/* This file was auto-generated by generate_css.py in jupyterlab_pygments */
"""


def main(style_class=JupyterStyle, output_path=None):
    """
    生成 CSS 文件。
    
    Parameters
    ----------
    style_class : pygments.style.Style
        要使用的 Pygments 样式类，默认为 JupyterStyle
    output_path : Path or None
        输出文件路径，默认为 style/base.css
    """
    formatter = HtmlFormatter(style=style_class)
    css = formatter.get_style_defs('.highlight')
    highlight_css = '\n'.join(filter(
        lambda line: line.startswith('.highlight'), css.splitlines()
    ))

    here = Path(__file__).parent
    style_file = output_path or (here / "style" / "base.css")
    style_file.write_text("\n".join((prefix, highlight_css)))
    
    print(f"CSS 已生成: {style_file}")
    print(f"使用样式类: {style_class.__name__}")


if __name__ == "__main__":
    main()
```

运行：

```bash
python generate_css.py
```

## 示例 3：验证自定义样式的 CSS 输出

```python
"""验证自定义样式生成的 CSS 是否正确"""

from pygments.formatters import HtmlFormatter
from pygments.lexers import PythonLexer
from pygments import highlight

# 导入你的自定义样式
from jupyterlab_pygments import JupyterStyle

def inspect_css(style_class):
    """检查样式类生成的 CSS 规则"""
    formatter = HtmlFormatter(style=style_class)
    css = formatter.get_style_defs('.highlight')
    
    print(f"=== {style_class.__name__} 生成的 CSS 规则 ===")
    for line in css.splitlines():
        if line.startswith('.highlight'):
            print(f"  {line}")
    
    # 统计规则数量
    rules = [l for l in css.splitlines() if l.startswith('.highlight')]
    print(f"\n共 {len(rules)} 条 .highlight 规则")

def test_highlight(style_class):
    """用一段 Python 代码测试高亮效果"""
    code = '''
@decorator
class MyClass:
    """A simple class."""
    def __init__(self, value=42):
        self.value = value  # inline comment
    
    def greet(self, name):
        return f"Hello, {name}!"

result = MyClass()
print(result.greet("World"))
'''
    formatter = HtmlFormatter(style=style_class)
    html = highlight(code, PythonLexer(), formatter)
    return html

if __name__ == "__main__":
    inspect_css(JupyterStyle)
    print()
    html = test_highlight(JupyterStyle)
    print("HTML 输出片段（前 500 字符）：")
    print(html[:500])
```

## 重新构建扩展

修改样式并重新生成 CSS 后，需要重建 JupyterLab 扩展：

```bash
# 步骤 1：重新生成 CSS
python generate_css.py

# 步骤 2：编译 TypeScript
jlpm build:lib

# 步骤 3：构建 labextension（开发模式）
jlpm build:labextension:dev

# 步骤 4：如果是可编辑安装，直接刷新 JupyterLab 即可看到效果
# 如果是正式安装，需要重新 pip install
pip install -e .
```

然后在 JupyterLab 中切换主题（Settings → JupyterLab Theme），观察语法高亮颜色是否跟随主题变化。
