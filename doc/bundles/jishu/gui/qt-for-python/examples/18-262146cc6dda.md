在 Qt 显示图片是一个很常见的需求，本文主要介绍一些显示图片的方法。

## QLabel 显示图片

```python
from xinet import QtWidgets, QtCore, QtGui
from xinet.run_qt import run

class Image(QtWidgets.QWidget):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.lbl = QtWidgets.QLabel()
        pixmap = QtGui.QPixmap("w.jpg")
        self.lbl.setPixmap(pixmap)  # 在 label 上显示图片
        # 图片随着标签大小而变化
        self.lbl.setScaledContents(True)  # 让图片自适应label大小
        self.initUI()

    def initUI(self):
        hbox = QtWidgets.QHBoxLayout()
        hbox.addWidget(self.lbl)
        self.setLayout(hbox)
        self.move(300, 200)
        self.setWindowTitle('自适应显示图片')
        self.show()

if __name__ == '__main__':
    run(Image)
```

self.lbl.setPixmap(QPixmap(""))   # 移除label上的图片

## QLabel 显示 gif 动图

```python
from xinet import QtWidgets, QtCore, QtGui
from xinet.run_qt import run

class Image(QtWidgets.QWidget):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.label = QtWidgets.QLabel()
        self.label.setGeometry(QtCore.QRect(0, 0, 500, 300))
        self.gif = QtGui.QMovie('a.gif')
        self.label.setMovie(self.gif)
        self.gif.start()  # 开始播放GIF动画
        self.initUI()

    def initUI(self):
        hbox = QtWidgets.QHBoxLayout()
        hbox.addWidget(self.label)
        self.setLayout(hbox)
        self.move(300, 200)
        self.setWindowTitle('自适应显示图片')
        # 设置窗体无边框
        self.setWindowFlags(QtCore.Qt.FramelessWindowHint)
        # 设置背景透明
        self.setAttribute(QtCore.Qt.WA_TranslucentBackground)

if __name__ == '__main__':
    run(Image)
```
